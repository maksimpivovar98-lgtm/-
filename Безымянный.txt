<script>
    // --- Состояние игры ---
    const state = {
        screen: 'menu', // menu, controls, game
        mode: '1p', // 1p или 2p
        score: 0,
        gameOver: false,
        player1Pos: 15, // Процент от левого края дороги
        player2Pos: 15, // Процент от правого края дороги (для врага)
        speed: 5,
        obstacles: [],
        keys: {
            ArrowLeft: false,
            ArrowRight: false,
            KeyA: false,
            KeyD: false
        },
        animationId: null,
        lastTime: 0
    };

    // --- DOM Элементы ---
    const menuScreen = document.getElementById('menu-screen');
    const controlsScreen = document.getElementById('controls-screen');
    const gameScreen = document.getElementById('game-screen');
    const btn1p = document.getElementById('btn-1p');
    const btn2p = document.getElementById('btn-2p');
    const btnStartGame = document.getElementById('btn-start-game');
    const btnExit = document.getElementById('btn-exit');
    const car1 = document.getElementById('car-1');
    const car2 = document.getElementById('car-2');
    const label2p = document.getElementById('label-2p');
    const rightControls = document.getElementById('right-controls');
    const scoreSpan = document.getElementById('score');
    const road = document.querySelector('.road');

    // --- Инициализация и переключение экранов ---
    function showScreen(screenName) {
        state.screen = screenName;
        menuScreen.classList.add('hidden');
        controlsScreen.classList.add('hidden');
        gameScreen.classList.add('hidden');

        if (screenName === 'menu') menuScreen.classList.remove('hidden');
        if (screenName === 'controls') controlsScreen.classList.remove('hidden');
        if (screenName === 'game') gameScreen.classList.remove('hidden');
    }

    btn1p.addEventListener('click', () => {
        state.mode = '1p';
        label2p.textContent = 'BOT';
        rightControls.style.display = 'none'; // Скрываем кнопки для второго игрока в режиме бота
        showScreen('controls');
    });

    btn2p.addEventListener('click', () => {
        state.mode = '2p';
        label2p.textContent = '2P';
        rightControls.style.display = 'flex'; // Показываем кнопки для второго игрока
        showScreen('controls');
    });

    btnStartGame.addEventListener('click', () => {
        startGame();
    });

    btnExit.addEventListener('click', () => {
        stopGame();
        showScreen('menu');
    });

    // --- Логика игры ---
    function startGame() {
        state.score = 0;
        state.gameOver = false;
        state.player1Pos = 15;
        state.player2Pos = 15;
        state.speed = 5;
        state.obstacles = [];
        state.keys = { ArrowLeft: false, ArrowRight: false, KeyA: false, KeyD: false };
        
        scoreSpan.textContent = '0';
        updateCarPositions();
        
        // Очищаем старые препятствия
        document.querySelectorAll('.obstacle').forEach(el => el.remove());

        showScreen('game');
        
        // Запускаем игровой цикл
        state.lastTime = performance.now();
        state.animationId = requestAnimationFrame(gameLoop);
        
        // Спавн препятствий
        spawnObstacle();
    }

    function stopGame() {
        state.gameOver = true;
        if (state.animationId) {
            cancelAnimationFrame(state.animationId);
        }
    }

    function updateCarPositions() {
        // player1Pos - это % от левого края дороги (0-100)
        // Но дорога занимает 60% ширины экрана, начиная с 20%.
        // Машина должна двигаться в пределах дороги.
        // Дорога: от 20% до 80% ширины контейнера.
        // Машина 1: left = 20% + (player1Pos / 100) * 60% - (ширина машины / 2)
        // Упростим: будем задавать left в процентах от контейнера.
        
        const roadLeft = 20; // 20%
        const roadWidth = 60; // 60%
        const carWidthPercent = 13; // ~60px от 450px

        // Для игрока 1
        let p1Left = roadLeft + (state.player1Pos / 100) * (roadWidth - carWidthPercent);
        car1.style.left = p1Left + '%';

        // Для игрока 2 / бота
        // player2Pos - это % от правого края дороги.
        // Значит, от левого края дороги это (100 - player2Pos).
        let p2LeftRoad = 100 - state.player2Pos; 
        let p2Left = roadLeft + (p2LeftRoad / 100) * (roadWidth - carWidthPercent) - carWidthPercent;
        // Корректировка, чтобы не вылетал за правый край
        if (p2Left + carWidthPercent > 80) p2Left = 80 - carWidthPercent;
        if (p2Left < 20) p2Left = 20;
        
        car2.style.left = p2Left + '%';
        
        // Сохраняем реальные координаты для проверки столкновений
        car1.dataset.left = p1Left;
        car2.dataset.left = p2Left;
    }

    // --- Ввод ---
    document.addEventListener('keydown', (e) => {
        if (state.screen !== 'game') return;
        if (e.code === 'ArrowLeft') state.keys.ArrowLeft = true;
        if (e.code === 'ArrowRight') state.keys.ArrowRight = true;
        if (e.code === 'KeyA') state.keys.KeyA = true;
        if (e.code === 'KeyD') state.keys.KeyD = true;
        // Предотвращаем прокрутку страницы стрелками
        if (['ArrowLeft', 'ArrowRight', 'ArrowUp', 'ArrowDown'].includes(e.code)) {
            e.preventDefault();
        }
    });

    document.addEventListener('keyup', (e) => {
        if (e.code === 'ArrowLeft') state.keys.ArrowLeft = false;
        if (e.code === 'ArrowRight') state.keys.ArrowRight = false;
        if (e.code === 'KeyA') state.keys.KeyA = false;
        if (e.code === 'KeyD') state.keys.KeyD = false;
    });

    // Сенсорное управление
    function setupTouchControls() {

        const btnRight1 = document.getElementById('btn-right-1');
        const btnLeft2 = document.getElementById('btn-left-2');
        const btnRight2 = document.getElementById('btn-right-2');

        // Игрок 1
        btnLeft1.addEventListener('touchstart', (e) => { e.preventDefault(); state.keys.KeyA = true; });
        btnLeft1.addEventListener('touchend', (e) => { e.preventDefault(); state.keys.KeyA = false; });
        btnRight1.addEventListener('touchstart', (e) => { e.preventDefault(); state.keys.KeyD = true; });
        btnRight1.addEventListener('touchend', (e) => { e.preventDefault(); state.keys.KeyD = false; });

        // Игрок 2 (только если режим 2p)
        btnLeft2.addEventListener('touchstart', (e) => { 
            if (state.mode === '2p') { e.preventDefault(); state.keys.ArrowLeft = true; } 
        });
        btnLeft2.addEventListener('touchend', (e) => { 
            if (state.mode === '2p') { e.preventDefault(); state.keys.ArrowLeft = false; } 
        });
        btnRight2.addEventListener('touchstart', (e) => { 
            if (state.mode === '2p') { e.preventDefault(); state.keys.ArrowRight = true; } 
        });
        btnRight2.addEventListener('touchend', (e) => { 
            if (state.mode === '2p') { e.preventDefault(); state.keys.ArrowRight = false; } 
        });
    }
    setupTouchControls();

    // --- Спавн препятствий ---
    function spawnObstacle() {
        if (state.gameOver || state.screen !== 'game') return;

        const obstacle = document.createElement('div');
        obstacle.className = 'car car-enemy obstacle'; // Используем стиль машины, но добавим класс obstacle
        obstacle.style.width = '50px';
        obstacle.style.height = '80px';
        obstacle.style.borderRadius = '8px';
        obstacle.style.position = 'absolute';
        obstacle.style.top = '-100px';
        obstacle.style.zIndex = '1';
        obstacle.style.boxShadow = '0 5px 10px rgba(0,0,0,0.5)';
        
        // Случайный цвет
        const colors = ['#f4a261', '#e76f51', '#2a9d8f', '#e9c46a', '#9b5de5'];
        obstacle.style.backgroundColor = colors[Math.floor(Math.random() * colors.length)];

        // Случайная позиция по ширине дороги (от 20% до 80% контейнера)
        const minLeft = 20;
        const maxLeft = 75; // 80% - ширина машины
        const leftPos = Math.random() * (maxLeft - minLeft) + minLeft;
        obstacle.style.left = leftPos + '%';

        road.appendChild(obstacle);
        
        // Сохраняем в массив
        state.obstacles.push({
            element: obstacle,
            left: leftPos,
            top: -100,
            width: 50,
            height: 80
        });

        // Следующее препятствие через случайное время
        const nextSpawn = Math.max(400, 1200 - state.score * 10);
        setTimeout(spawnObstacle, nextSpawn);
    }

    // --- Игровой цикл ---
    function gameLoop(timestamp) {
        if (state.gameOver || state.screen !== 'game') return;

        const deltaTime = timestamp - state.lastTime;
        state.lastTime = timestamp;

        // Управление игроком 1 (A/D или стрелки)
        if (state.keys.KeyA || state.keys.ArrowLeft) {
            state.player1Pos = Math.max(0, state.player1Pos - 1.5);
        }
        if (state.keys.KeyD || state.keys.ArrowRight) {
            state.player1Pos = Math.min(100, state.player1Pos + 1.5);
        }

        // Управление игроком 2 / Ботом
        if (state.mode === '2p') {
            // Игрок 2 управляет стрелками (левая/правая)
            // Но мы уже используем ArrowLeft/Right для игрока 1.
            // Нужно разделить управление. 
            // Для 2P: Игрок 1 - A/D, Игрок 2 - Стрелки.
            // Перепишем логику выше.
        } else {
            // Режим Бота
            // Бот просто двигается в случайную сторону или следует за игроком
            // Простая логика: бот пытается избегать препятствий или просто смещается
            if (Math.random() < 0.02) {
                state.player2Pos += (Math.random() - 0.5) * 10;
            }
            // Ограничим
            state.player2Pos = Math.max(0, Math.min(100, state.player2Pos));
        }

        // Пересчет управления для 2P (если нужно)
        // В текущей реализации для 2P: 
        // Игрок 1: A/D
        // Игрок 2: ArrowLeft/ArrowRight
        if (state.mode === '2p') {
            // Сброс управления игрока 1, если использовались стрелки
            // Но мы используем KeyA/KeyD для 1P и Arrow для 2P.
            // Переопределим:
            // Игрок 1: KeyA / KeyD
            // Игрок 2: ArrowLeft / ArrowRight
            if (state.keys.KeyA) state.player1Pos = Math.max(0, state.player1Pos - 1.5);
            if (state.keys.KeyD) state.player1Pos = Math.min(100, state.player1Pos + 1.5);
            
            if (state.keys.ArrowLeft) state.player2Pos = Math.min(100, state.player2Pos + 1.5); // + потому что от правого края
            if (state.keys.ArrowRight) state.player2Pos = Math.max(0, state.player2Pos - 1.5);
        } else {
            // Режим 1P: Игрок 1 управляет A/D или стрелками
            if (state.keys.KeyA || state.keys.ArrowLeft) state.player1Pos = Math.max(0, state.player1Pos - 1.5);
            if (state.keys.KeyD || state.keys.ArrowRight) state.player1Pos = Math.min(100, state.player1Pos + 1.5);
            
            // Бот
            // Бот немного тупит, но иногда уворачивается
            if (Math.random() < 0.05) {
                state.player2Pos += (Math.random() - 0.5) * 15;
                state.player2Pos = Math.max(0, Math.min(100, state.player2Pos));
            }
        }

        updateCarPositions();

        // Движение препятствий
        const speed = 5 + (state.score * 0.2); // Ускорение
        for (let i = state.obstacles.length - 1; i >= 0; i--) {
            const obs = state.obstacles[i];
            obs.top += speed;
            obs.element.style.top = obs.top + 'px';

            // Проверка столкновения с игроком 1
            const car1Rect = car1.getBoundingClientRect();
            const obsRect = obs.element.getBoundingClientRect();
            
            // Простая проверка на пересечение прямоугольников
            if (
                car1Rect.left < obsRect.right &&
                car1Rect.right > obsRect.left &&
                car1Rect.top < obsRect.bottom &&
                car1Rect.bottom > obsRect.top
            ) {
                // Столкновение!
                stopGame();
                alert('Игра окончена! Ваш счет: ' + state.score);
                showScreen('menu');
                return;
            }

            // Проверка столкновения с игроком 2 (если 2P)
            if (state.mode === '2p') {
                const car2Rect = car2.getBoundingClientRect();
                if (
                    car2Rect.left < obsRect.right &&
                    car2Rect.right > obsRect.left &&
                    car2Rect.top < obsRect.bottom &&
                    car2Rect.bottom > obsRect.top
                ) {
                    stopGame();
                    alert('Игрок 2 проиграл! Ваш счет: ' + state.score);
                    showScreen('menu');
                    return;
                }
            }

            // Удаление препятствия, если оно ушло за экран
            if (obs.top > gameScreen.clientHeight) {
                obs.element.remove();
                state.obstacles.splice(i, 1);
                state.score += 10;
                scoreSpan.textContent = state.score;
            }
        }

        state.animationId = requestAnimationFrame(gameLoop);
    }

    // Начальная инициализация
    showScreen('menu');

</script>
